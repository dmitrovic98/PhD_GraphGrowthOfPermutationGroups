/* SemiPrimitive = Every normal subgroup is transitive or semiregular. Equivalently, one can check the kernels of block systems for semiregularity */

IsSemiPrimitive := function(L)
   AllParts := AllPartitions(L);
   for B in AllParts do
    	N := BlocksKernel(L,B);
    	if not IsSemiregular(N) then return false; end if;
   end for;
   return true;
end function;

/********************************************************************************* Regular Normal Nilpotent Subgroup *************************************************************************/

/* Theorem 1.1 (Guidici, Morgan - Locally Semiprimitive graphs and a Theorem of Weiss) */
/* Let L be a semiprimitive permutation group of degree d. If (d,6)=1 and L contains a regular normal nilpotent subgroup, then the Graph Growth of L is constant. */

IsNilSubGrpRes := function(L)
	x := GCD(Degree(L),6);
	if IsSemiPrimitive(L) and (x eq 1) then
		N := [ Cl`subgroup : Cl in NormalSubgroups(L : IsRegular, IsNilpotent)];
		return (#N gt 0);
	else
		return false;
	end if;
end function;

/* Corollary 1.3 (Guidici, Morgan - Locally Semiprimitive graphs and a Theorem of Weiss) */
/* Let L be a semiprimitive permutation group. Let R be a normal regular nilpotent subgroup of L.*/
/* Then there exist normal subgroups F < R < J such that J/F = S3^n or A4^m (for some integers n,m), */
/* OTHERWISE, L is Graph-Restrictive. */

IsGaPowerOfH:= function(G, H)
	n := Order(G);
	m := Order(H);
	k := Log(m,n);
	if IsIntegral(k) then
		GG := DirectProduct([ H : i in [1..k] ]);
		b, _ := IsIsomorphic(G,GG);
		return b;
	else
		return false;
	end if;
end function;

IsFRJNilpotentRes := function(L)
	if IsSemiPrimitive(L) then
		AllNormal := [ Cl`subgroup : Cl in NormalSubgroups(L) ];
		ListOfRs := [ R : R in AllNormal | IsRegular(R) and IsNilpotent(R) ];
		if (#ListOfRs eq 0) then return false; end if;
		S3 := Sym(3);
		A4 := Alt(4);
		for R in ListOfRs do
			ListOfFs := [ F : F in AllNormal | (F subset R) and (Order(F) lt Order(R)) ];
			ListOfJs := [ J : J in AllNormal | (R subset J) and (Order(R) lt Order(J)) ];
			for F in ListOfFs do
				for J in ListOfJs do
					G := J/F;
					if IsGaPowerOfH(G,S3) or IsGaPowerOfH(G,A4) then
						return false;
					end if;
				end for;
			end for;
		end for;
		return true;
	else
		return false;
	end if;
end function;

/* Let L be a transitive permutation group. If there exist integers n, m <= n/2 and a prime power q such that L contains a normal subgroup (permutation) isomorphic to PSL(n,q) in its natural action on the set of m-subspaces of Fq^n, then the Graph Growth of L is Constant. */

/* Let n >= 2, 1 <= m <= n/2 be integers and q a prime power. Let L(n,m,q) denote the permutation group induced by the action of PSL(n,q) on the m-subspaces of F_q^n. The following are precisely the values of parameters (n,m,q) such that the degree of L(n,m,q) is at most 48. */

/* If m = 1, then L(n,m,q) is permutation isomorphic to PSL(n,q) in its natural action on projective points, which is 2-transitive! Hence, the Graph Growth of L(n,m,q), as well as any group containing it, is Constant by the virtue of being 2-transitive. */
/* Therefore, the only genuinely new case is given by (4,2,2) of degree 35. The index of L(4,2,2) in its normalizer in Sym(35) is 2. The following have Constant Graph Growth. */
/* TG(35,36) = L(4,2,2) (abstractly isomorphic to PSL(4,2) = Alt(8)) */
/* TG(35,44) = N_{Sym(35)}(L(4,2,2)) (abstractly isomorphic to Sym(8)). */

/* The following returns the group L(n,m,q) */

PSLOnSubSpaces := function(n,m,q)
	G, Gen := PSL(n,q);
	V := VectorSpace(GF(q),n);
	mSubSpacesGSet := {@ @};
	mSubSpaces := {@ @};
	MGen := Subsets(Set(Gen),m);

	for S in MGen do
		SS := sub<V|S>;
		if not (Dimension(SS) eq m) then continue; end if;
		found := false;
		if (#mSubSpaces eq 0) then
			mSubSpaces := mSubSpaces join {SS};
			mSubSpacesGSet := mSubSpacesGSet join {{Index(Gen,v): v in S}};
		else
			for U in mSubSpaces do
				if (U eq SS) then
					i := Index(mSubSpaces,U);
					found := true;
					break;
				end if;
			end for;
			if found then
				mSubSpacesGSet := {@ mSubSpacesGSet[j] : j in [1..(i-1)] @} join {@ mSubSpacesGSet[i] join {Index(Gen,v): v in S} @} join {@ mSubSpacesGSet[j] : j in [(i+1)..(#mSubSpacesGSet)] @};
			else
				mSubSpaces := mSubSpaces join {SS};
				mSubSpacesGSet := mSubSpacesGSet join {{Index(Gen,v): v in S}};
			end if;
		end if;
	end for;

	print mSubSpacesGSet, #mSubSpacesGSet;
	L := ActionImage(G,GSet(G,mSubSpacesGSet));
	return L;

end function;

end function;

/* Theorem 1.1 (Morgan - A Class of Semiprimitive groups that are graph-restrictive) */
/* Let L be the Frobenius group of order 2 * 3^n and degree 3^n with elementary abelian Frobenius kernel. Then L is graph-restrictive */
/* Remark: There is no command to find the Frobenius Kernel of a group, but it is a known fact that the Fitting Group of a Frobenius Group is precisely its Frobenius Kernel */

IsFrobeniusRes := function(L)
	if not ((#L mod 2) eq 0) then return false; end if;
	d := #L div 2;
	b, p, k := IsPrimePower(d);
	if b then
		deg := Degree(L);
		if (p eq 3) and (deg eq p^k) then 
			if IsFrobenius(L) then
				K := FittingGroup(L);
				if IsElementaryAbelian(K) then 
					return true;
				else
					return false;
				end if;
			else
				return false;
			end if;
		else
			return false;
		end if;
	else
		return false;
	end if;
end function;

/********************************************************************* (WEAKLY) p-SUB-REGULAR GROUPS *************************************************************************************************/

/* A permutation group L is p-sub-regular if it is transitive and there exist two points x and y such that L is generated by Lx and Ly and the action of Lx on Y := y^(Lx) is permutation isomorphic to Cp */
/* Equivalently */
/* A permutation group L is p-sub-regular if it is transitive, Lv is cyclic of order p and there exist two points x and y such that Lx and Ly generate the full group L */
/* Verret has shown that p-sub-regular groups have Constant Graph Growth */

IsPSubRegular := function(L)
	if not IsTransitive(L) then return false; end if;
	deg := Degree(L);
	Lx := Stabiliser(L,1);
	p := Order(Lx);
	if not IsPrime(p) then return false; end if;
	for y in [2..deg] do
		Ly := Stabiliser(L,y);
		LL := sub<L|Lx,Ly>;
		if L eq LL then return true; end if;
	end for;
	return false;
end function;

/* A permutation group L is weakly p-sub-regular if it is transitive, Lv is cyclic of order p and there exist two points x and y such that if LL is the group generated by Lx and Ly, then every z is in LL-orbit of x or LL-orbit of y */

IsWeaklyPSubRegular := function(L)
    	if not IsTransitive(L) then return false; end if;
	deg := Degree(L);
  	Lx := Stabilizer(L,1);
    	p := Order(Lx);
   	if not IsPrime(p) then return false; end if;
    	for y in [2..deg] do
        	Ly := Stabilizer(L,y);
        	LL := sub<L| Lx,Ly >;
        	Ox := 1^LL;
        	Oy := y^LL;
        	O := Ox join Oy;
        	if #O eq deg then return true; end if;
    	end for;
    	return false;
end function;

/********************************************************************************* p-graph-restrictive **********************************************************************************************/

/* A permutation group L is p-graph-restrictive if it is transitive and there exists a constant c := c(p,L) such that for all locally-L pairs (X,G), the largest power of p dividing the order of a stabiliser of an arc of X in G is at most c */


/* Corollary 10 (Potočnik, Spiga, Verret - On graph-restrictive permutation groups) */
/* Let L be a semiprimitive group acting on a set X, let x in X and let p be a prime. If Op(Lx)= 1, then L is p-graph-restrictive. */

Op_pRes := function(L,p)
    	Lx := Stabilizer(L,1);
	if IsSemiPrimitive(L) and (Order(pCore(Lx,p)) eq 1) then
        	return true;
	end if;
    return false;
end function;

/* Corollary 11 (PSV Graph Restrictive article */
/* Let p be a prime and let L be a transitive permutation group on a set X. Let x in X and let P be a Sylow p-subgroup of Lx. If
(1) |P| = p, and
(2) there exists l in L such that <P, P^l> is transitive on X,
then L is p-graph-restrictive. */

PPl_pRes := function(L,p)
	Lx := Stabiliser(L,1);
    	P := Sylow(Lx, p);
    	if not (Order(P) eq p) then return false; end if;
    	for l in L do
        	PPl := sub<L|P,P^l>;
        	if IsTransitive(PPl) then return true; end if;
    	end for;
    	return false;
end function;

/* Observation: A transitive group L is graph-restrictive if and only if it is p-graph restrictive for all primes p dividing the order of the stabiliser Lx */

IspRes := function(L,p);
	return (Op_pRes(L,p) or PPl_pRes(L,p));
end function;

IspResAllp := function(L);
	Lx := Stabiliser(L,1);
	P := PrimeDivisors(#Lx);
	for p in P do
		if not IspRes(L,p) then return false; end if;
	end for;
	return true;
end function;

/******************************************************************************* TWO BLOCKS THEOREMS *************************************************************************************************/

/* Corollary 9 (Potočnik, Spiga, Verret - ON THE ORDER OF ARC-STABILISERS IN ARC-TRANSITIVE GRAPHS WITH PRESCRIBED LOCAL GROUP)) */
/* Let L be a transitive NON-REGULAR (i.e., not semiregular) permutation group admitting a block system with two blocks A and B */
/* If the pointwise stabiliser L(A) of the block A is non-trivial, the graph growth is exponential */
/* Otherwise, the graph growth of L is AT LEAST polynomial */

TwoBlocks := function(L)
	if IsRegular(L) then return false, 0; end if;
	deg := Degree(L);
	Parts := [x : x in AllPartitions(L)];
	BigBlocks := [B : B in Parts | 2 * #B eq deg];
	if (#BigBlocks eq 0) then
		return false, 0;
	else
		for B in BigBlocks do
			LBsetwise := Stabiliser(L,B);
			LBpointwise := ActionKernel(LBsetwise,GSet(LBsetwise,B));
			if (#LBpointwise gt 1) then 
				return true, #LBpointwise;
			end if;
		end for;
	end if;
	return true, 1;
end function;

/********************************************************************** Thesis: BLOCK COMPLEMENT THEOREM ***********************************************************************************************/

/* Let L <= Sym(X) be a transitive permutation group. If B is a PROPER block for L and L_(X\B) neq 1, then the Graph Growth of L is Exponential */
/* Note that if C subset B, then X\B subset X\C and L_(X\C) <= L_(X\B) */
/* Hence, BCT applies to L with block B if and only if it is applies to L and some maximal block B */

BlockComplementTheorem := function(L)
	AllBlocks := AllPartitions(L);
	MaxBlocks :=  [ B : B in AllBlocks | not exists{ C : C in AllBlocks | B subset C and B ne C } ];
	for B in MaxBlocks do
		BComp := [i : i in GSet(L) | not i in B];
		PointWiseStabOfBComp := L;
		for i in BComp do
			PointWiseStabOfBComp := PointWiseStabOfBComp  meet Stabiliser(L,i);
			if IsTrivial(PointWiseStabOfBComp) then
				break;
			end if;
		end for;
		if not IsTrivial(PointWiseStabOfBComp) then return true; end if;
	end for;
	return false;
end function;

/******************************************************************** Thesis: Sum-zero Permutation Groups **********************************************************************************************/

/* Construting sum-zero groups between Sigma0(K,N) and K wr G where K is abelian with exponent 2 */

SumZeroExp2 := function(k,N,G)

	Ab := AbelianGroup([2 : i in [1..k]]);
	rho, RegAb := RegularRepresentation(Ab, sub<Ab|>);
	Gens := [rho(Ab.i) : i in [1..k]];
	W, ListOfAlphas, beta, _ := WreathProduct(RegAb,G);
	SumZeroGens := [ ListOfAlphas[1](g) * ListOfAlphas[j](g) : g in Gens, j in [2..n] ];
	Sigma0 := sub<W|SumZeroGens>;
	Sigma0N := sub<W|Sigma0,beta(N)>;
	return [Sigma0N] cat IntermediateSubgroups(W,Sigma0N) cat [W];
end function;

GraphGrowthConjectures := function(L)
	if IsSemiPrimitive(L) then print "L is semiprimitive"; print "Constant Growth by the PSV Conjecture"; return ""; end if;
	Blocks := [x : x in AllPartitions(L)];
	for B in Blocks do
		BB := B^L;
		K := BlocksKernel(L,BB);
		KB := ActionKernel(K,GSet(K,B));
		if (#KB gt 1) then
			print "Kernel pointwise stabiliser of the following block is non-trivial"; print B;
			print "Exponential graph growth by ExpConjecture";
			return "";
		end if;
	end for;
	print "We conjecture graph growth of L is Polynomial";
	return "";
end function;

ExpConj := function(L)
	Blocks := AllPartitions(L);
	if #Blocks eq 0 then return false; end if;
	for B in Blocks do
		BB := B^L;
		K := BlocksKernel(L,BB);
		KB := ActionKernel(K,GSet(K,B));
		if #KB gt 1 then return true; end if;
	end for;
	return false;
end function;
